# SPEC 11 and 12: checking proofs. Each case checks a proof document against
# a program and expects the exact report. Failure details are free text.

=== a valid proof is checked
spec: 11, 12
args: --check-proof proof.py program.py
--- program.py
from peye import *
fact(p('a'))
forward(q(X), p(X))
forward(r(X), q(X))
--- proof.py
q('a')
r('a')

clause(1, fact(p('a')))
clause(2, forward(q(X), p(X)))
clause(3, forward(r(X), q(X)))

step(q('a'), clause(2), {'X': 'a'}, [p('a')])
step(p('a'), clause(1), {}, [])
step(r('a'), clause(3), {'X': 'a'}, [q('a')])
--- stdout
condition('C1', 'resolution', 'ok', 3)
condition('C2', 'well_founded', 'ok', 3)
condition('C3', 'justification', 'ok', 3)
condition('C4', 'coverage', 'ok', 4)
condition('C5', 're_decision', 'ok', 0)
condition('C6', 'boundary_consistency', 'ok', 0)
condition('C7', 'relevance', 'ok', 5)
steps(3)
verified(3)
recomputed(0)
composed(0)
trusted(0)
claims(2)
verdict('checked')
