# SPEC 9: reading documents. Accepted forms are read through facts_from and
# printed back; rejected forms are given as goals and must stop the run.

=== canonical text reads back as the same term
spec: 8, 9
--- program.py
from peye import *
facts_from('data.py')
query(d(X))
--- data.py
d('plain')
d("it's")
d([1, [2, 3], *T])
d(struct('hello world', 1))
d(1 + 2 * 3)
d((1 + 2) * 3)
d(2 ** -1)
d(-2 ** 2)
d((-2) ** 2)
d(struct('-', 1))
d(--'x')
d('a' | 'b' & 'c')
d((1 < 2) & (3 >= 4))
d(1e+22)
d(-0.5)
--- stdout
d('plain')
d("it's")
d([1, [2, 3], *…])
d(struct('hello world', 1))
d(1 + 2 * 3)
d((1 + 2) * 3)
d(2 ** -1)
d(-2 ** 2)
d((-2) ** 2)
d(struct('-', 1))
d(--'x')
d('a' | 'b' & 'c')
d((1 < 2) & (3 >= 4))
d(1e+22)
d(-0.5)

=== a name reads as a variable, and each _ as a new one
spec: 9
args: program.py --goal "same(1, 1)" --goal "same(1, 2)" --goal "any(1, 2)"
--- program.py
from peye import *
facts_from('data.py')
--- data.py
same(X, X)
any(_, _)
--- stdout
same(1, 1)
any(1, 2)
